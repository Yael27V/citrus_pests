📄 Resumen Ejecutivo

El Problema

Los agricultores cítricultores enfrentan pérdidas del 30-40% de sus cosechas por plagas sin contar con herramientas digitales modernas para:

  • Identificación temprana y precisa de plagas
  • Monitoreo eficiente en múltiples ranchos
  • Acceso a información técnica confiable y actualizada
  • Documentación sistemática de plagas detectadas

La Solución

Plagas Beta es una plataforma web integrada que centraliza:

  • ✅ Catálogo completo de plagas con fichas técnicas detalladas
  • ✅ Monitoreo y registro por rancho/ubicación agrícola
  • ✅ Galería de imágenes de referencia (campo y microscopio)
  • ✅ Registro histórico de detecciones y severidad
  • ✅ Guías de manejo y recomendaciones

Ejemplo: Hook useEffect para Cargar Datos

export default function PlagasPage() {
  const [plagas, setPlagas] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.getPlagas()
      .then(data => setPlagas(data))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <LoadingSpinner />;

  return (
    <div className="grid md:grid-cols-3 gap-4">
      {plagas.map(plaga => (
        <PlagaCard key={plaga.id} plaga={plaga} />
      ))}
    </div>
  );
}

Rutas de Plagas (server/routes/plagas.js)

import express from 'express';
import { createClient } from '@supabase/supabase-js';

const router = express.Router();
const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY
);

// GET /api/plagas
router.get('/plagas', async (req, res) => {
  try {
    const { data, error } = await supabase
      .from('plagas')
      .select('*');

    if (error) throw error;
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// GET /api/plagas/:id
router.get('/plagas/:id', async (req, res) => {
  try {
    const { data, error } = await supabase
      .from('plagas')
      .select('*')
      .eq('id', req.params.id)
      .single();

    if (error) throw error;
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

export default router;

Base de Datos - Esquema SQL

Tabla PLAGAS

CREATE TABLE plagas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre VARCHAR(255) NOT NULL UNIQUE,
  nombre_cientifico VARCHAR(255),
  tipo VARCHAR(100),
  familia VARCHAR(100),
  orden VARCHAR(100),
  descripcion TEXT,
  ciclo_vida TEXT,
  sintomas TEXT,
  danos TEXT,
  imagen_url TEXT,
  imagenes_adicionales TEXT[],
  guia_monitoreo TEXT,
  umbral_accion VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_plagas_nombre ON plagas(nombre);

Tabla RANCHOS

CREATE TABLE ranchos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre VARCHAR(255) NOT NULL,
  ubicacion TEXT,
  cultivo_principal VARCHAR(100),
  hectareas DECIMAL(10, 2),
  propietario VARCHAR(255),
  telefono VARCHAR(20),
  email VARCHAR(255),
  latitud DECIMAL(10, 8),
  longitud DECIMAL(11, 8),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Tabla REGISTROS_PLAGAS

CREATE TABLE registros_plagas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  rancho_id UUID NOT NULL REFERENCES ranchos(id),
  plaga_id UUID NOT NULL REFERENCES plagas(id),
  fecha_deteccion DATE NOT NULL,
  nivel_severidad VARCHAR(50),
  notas TEXT,
  imagen_url TEXT,
  técnico_nombre VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_registros_rancho ON registros_plagas(rancho_id);
CREATE INDEX idx_registros_plaga ON registros_plagas(plaga_id);
